  <%
    int visitorIdGenerated = 0; // Will hold the generated Visitor ID
    boolean registrationSuccess = false; // flag to show Explore More button
    if(request.getParameter("submit") != null){
        String visitorIdInput = request.getParameter("visitor_id");
        String fullname = request.getParameter("fullname");
        String email = request.getParameter("email");
        String phone = request.getParameter("phone");
        String purpose = request.getParameter("purpose");
        String visitDate = request.getParameter("visitDate");

        Connection con = null;
        PreparedStatement ps = null;
        ResultSet rs = null;
        try{
            Class.forName("com.mysql.cj.jdbc.Driver");
            con = DriverManager.getConnection("jdbc:mysql://localhost:3307/sanctuary","root",""); 

            String sql = "INSERT INTO visitors(visitor_id, fullname, email, phone, purpose, visitDate) VALUES(?,?,?,?,?,?)";
            ps = con.prepareStatement(sql, Statement.RETURN_GENERATED_KEYS);

            ps.setString(1, visitorIdInput);
            ps.setString(2, fullname);
            ps.setString(3, email);
            ps.setString(4, phone);
            ps.setString(5, purpose);
            ps.setString(6, visitDate);

            int rows = ps.executeUpdate();
            if(rows > 0){
                rs = ps.getGeneratedKeys();
                if(rs.next()){
                    visitorIdGenerated = rs.getInt(1);
                } else {
                    visitorIdGenerated = Integer.parseInt(visitorIdInput);
                }
                registrationSuccess = true; // set flag to true
            }
        }catch(Exception e){
            out.println("<div class='msg' style='color:yellow;'>Error: "+e.getMessage()+"</div>");
        }finally{
            try{ if(rs!=null) rs.close(); if(ps!=null) ps.close(); if(con!=null) con.close(); }catch(Exception e){}
        }
    }
  %>
