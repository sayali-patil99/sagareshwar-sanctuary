            Class.forName("com.mysql.cj.jdbc.Driver");

            String sql = "INSERT INTO donors(visitor_id, name, contact_no, email, address, amount) VALUES(?, ?, ?, ?, ?, ?)";
            ps = conn.prepareStatement(sql);
            ps.setString(1, visitor_id);
            ps.setString(2, name);
            ps.setString(3, contact_no);
            ps.setString(4, email);
            ps.setString(5, address);
            ps.setString(6, amount);

            int i = ps.executeUpdate();
            if(i > 0) {
                msg = "<p class='message'>Donor data submitted successfully!</p>";

                msg = "<p class='error'>Error! Please try again.</p>";

            msg = "<p class='error'>Error: " + e.getMessage() + "</p>";
        } finally {

        }
