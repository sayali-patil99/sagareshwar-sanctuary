<%
String message = "";
boolean registrationSuccess = false;

if(request.getParameter("submit") != null){
    String staffId = request.getParameter("staff_id").trim();
    String name = request.getParameter("name").trim();
    String role = request.getParameter("role");
    String shift = request.getParameter("shift");
    String attendance = request.getParameter("attendance");

    if(staffId.isEmpty() || name.isEmpty() || role.isEmpty() || shift.isEmpty() || attendance.isEmpty()){
        message = "❌ Please fill all fields!";
    } else {
        Connection con = null;
        PreparedStatement ps = null;
        try {
            Class.forName("com.mysql.cj.jdbc.Driver");
            con = DriverManager.getConnection("jdbc:mysql://localhost:3307/sanctuary", "root", ""); // adjust password

            String sql = "INSERT INTO staff(staff_id, name, role, shift, attendance) VALUES(?,?,?,?,?)";
            ps = con.prepareStatement(sql);
            ps.setString(1, staffId);
            ps.setString(2, name);
            ps.setString(3, role);
            ps.setString(4, shift);
            ps.setString(5, attendance);

            int rows = ps.executeUpdate();
            if(rows > 0){
                message = "✅ Staff data submitted successfully!";
                registrationSuccess = true; // allow Explore More button
            } else {
                message = "❌ Failed to submit staff data!";
            }
        } catch(SQLIntegrityConstraintViolationException dup){
            message = "❌ Staff ID already exists!";
        } catch(Exception e){
            message = "⚠️ Error: " + e.getMessage();
        } finally {
            try{ if(ps!=null) ps.close(); if(con!=null) con.close(); } catch(Exception e){}
        }
    }
}
